cmake_minimum_required(VERSION 3.16)
project(uart_reliable_transport C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(reliable_core src/reliable/reliable_transport.c src/reliable/reliable_sender.c)
target_include_directories(reliable_core PUBLIC include)
add_library(host_link src/platform/host/host_link.c)
target_include_directories(host_link PUBLIC include src/platform/host)
add_executable(reliable_uart_sim src/platform/host/host_main.c)
target_link_libraries(reliable_uart_sim PRIVATE reliable_core host_link)
enable_testing()
add_executable(test_reliable_transport tests/test_reliable_transport.c)
target_link_libraries(test_reliable_transport PRIVATE reliable_core)
add_test(NAME reliable_transport_tests COMMAND test_reliable_transport)
add_executable(test_reliable_sender tests/test_reliable_sender.c)
target_link_libraries(test_reliable_sender PRIVATE reliable_core host_link)
add_test(NAME reliable_sender_tests COMMAND test_reliable_sender)
