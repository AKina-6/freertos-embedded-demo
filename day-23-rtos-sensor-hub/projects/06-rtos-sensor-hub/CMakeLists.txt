cmake_minimum_required(VERSION 3.16)
project(rtos_sensor_hub C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(sensor_hub_core src/app/sensor_hub.c)
target_include_directories(sensor_hub_core PUBLIC include)
add_executable(sensor_hub_sim src/platform/host/host_main.c)
target_link_libraries(sensor_hub_sim PRIVATE sensor_hub_core)
enable_testing()
add_executable(test_sensor_hub tests/test_sensor_hub.c)
target_link_libraries(test_sensor_hub PRIVATE sensor_hub_core)
add_test(NAME sensor_hub_tests COMMAND test_sensor_hub)
