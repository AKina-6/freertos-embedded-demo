cmake_minimum_required(VERSION 3.16)
project(task_notification_isr_demo C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(notification_core src/app/task_notification_core.c src/app/isr_event_processor.c)
target_include_directories(notification_core PUBLIC include)
add_executable(task_notification_sim src/platform/host/host_main.c)
target_link_libraries(task_notification_sim PRIVATE notification_core)
enable_testing()
add_executable(test_task_notification tests/test_task_notification.c)
target_link_libraries(test_task_notification PRIVATE notification_core)
add_test(NAME task_notification_tests COMMAND test_task_notification)
add_executable(test_event_processor tests/test_event_processor.c)
target_link_libraries(test_event_processor PRIVATE notification_core)
add_test(NAME event_processor_tests COMMAND test_event_processor)
