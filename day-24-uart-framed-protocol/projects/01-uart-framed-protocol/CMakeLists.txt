cmake_minimum_required(VERSION 3.16)
project(uart_framed_protocol C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(protocol_core src/protocol/protocol_frame.c src/protocol/stream_parser.c)
target_include_directories(protocol_core PUBLIC include)
add_executable(uart_protocol_sim src/platform/host/host_main.c)
target_link_libraries(uart_protocol_sim PRIVATE protocol_core)
enable_testing()
add_executable(test_protocol_frame tests/test_protocol_frame.c)
target_link_libraries(test_protocol_frame PRIVATE protocol_core)
add_test(NAME protocol_frame_tests COMMAND test_protocol_frame)
add_executable(test_stream_parser tests/test_stream_parser.c)
target_link_libraries(test_stream_parser PRIVATE protocol_core)
add_test(NAME stream_parser_tests COMMAND test_stream_parser)
