cmake_minimum_required(VERSION 3.16)
project(i2c_multidevice_bus C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT MSVC)
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(i2c_core src/i2c/i2c_bus.c src/i2c/i2c_registry.c src/i2c/i2c_manager.c)
target_include_directories(i2c_core PUBLIC include)
add_library(host_i2c src/platform/host/host_i2c.c)
target_include_directories(host_i2c PUBLIC include src/platform/host)
add_executable(i2c_multidevice_sim src/platform/host/host_main.c)
target_link_libraries(i2c_multidevice_sim PRIVATE i2c_core host_i2c)
enable_testing()
add_executable(test_registry tests/test_registry.c)
target_link_libraries(test_registry PRIVATE i2c_core)
add_test(NAME registry_tests COMMAND test_registry)
add_executable(test_manager tests/test_manager.c)
target_link_libraries(test_manager PRIVATE i2c_core host_i2c)
add_test(NAME manager_tests COMMAND test_manager)
