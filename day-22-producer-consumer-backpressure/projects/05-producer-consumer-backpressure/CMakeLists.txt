cmake_minimum_required(VERSION 3.16)
project(producer_consumer_backpressure C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT MSVC)
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(pc_core src/app/ring_buffer.c src/app/producer_consumer.c)
target_include_directories(pc_core PUBLIC include)
add_executable(producer_consumer_sim src/platform/host/host_main.c)
target_link_libraries(producer_consumer_sim PRIVATE pc_core)
enable_testing()
add_executable(test_ring_buffer tests/test_ring_buffer.c)
target_link_libraries(test_ring_buffer PRIVATE pc_core)
add_test(NAME ring_buffer_tests COMMAND test_ring_buffer)
add_executable(test_producer_consumer tests/test_producer_consumer.c)
target_link_libraries(test_producer_consumer PRIVATE pc_core)
add_test(NAME producer_consumer_tests COMMAND test_producer_consumer)
