cmake_minimum_required(VERSION 3.16)
project(spi_master_slave_protocol C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(spi_core src/spi/spi_protocol.c src/spi/spi_transaction.c src/spi/spi_slave_model.c)
target_include_directories(spi_core PUBLIC include)
add_library(host_spi src/platform/host/host_spi.c)
target_include_directories(host_spi PUBLIC include src/platform/host)
add_executable(spi_protocol_sim src/platform/host/host_main.c)
target_link_libraries(spi_protocol_sim PRIVATE spi_core host_spi)
enable_testing()
add_executable(test_protocol tests/test_protocol.c)
target_link_libraries(test_protocol PRIVATE spi_core)
add_test(NAME spi_protocol_tests COMMAND test_protocol)
add_executable(test_transaction_slave tests/test_transaction_slave.c)
target_link_libraries(test_transaction_slave PRIVATE spi_core host_spi)
add_test(NAME spi_transaction_slave_tests COMMAND test_transaction_slave)
