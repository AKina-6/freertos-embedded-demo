cmake_minimum_required(VERSION 3.16)
project(can_message_model C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT MSVC)
 add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(can_core src/can/can_model.c)
target_include_directories(can_core PUBLIC include)
add_executable(can_bus_sim src/platform/host/host_main.c)
target_link_libraries(can_bus_sim PRIVATE can_core)
enable_testing()
add_executable(test_can tests/test_can.c)
target_link_libraries(test_can PRIVATE can_core)
add_test(NAME can_model_tests COMMAND test_can)
